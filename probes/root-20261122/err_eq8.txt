inst✝⁶ : CommRing B

inst✝⁴ : Algebra R S
inst✝³ : Algebra R B
f g : A →ₐ[R] B
T : Type u
inst✝² : CommRing T
inst✝¹ : Algebra R T
inst✝ : Module.Flat R T
eLHS : T ⊗[R] (S ⊗[R] ↥(f ≟ₐ g)) ≃ₐ[T] T ⊗[R] S ⊗[T] ↥(map (AlgHom.id T T) f ≟ₐ map (AlgHom.id T T) g) :=
  auxEquivLHS f g T
eRHS : T ⊗[R] ↥(map (AlgHom.id S S) f ≟ₐ map (AlgHom.id S S) g) ≃ₐ[T]
  ↥(map (AlgHom.id (T ⊗[R] S) (T ⊗[R] S)) (map (AlgHom.id T T) f) ≟ₐ
      map (AlgHom.id (T ⊗[R] S) (T ⊗[R] S)) (map (AlgHom.id T T) g)) :=
  auxEquivRHS f g T
this : IsScalarTower R T (T ⊗[R] ↥(map (AlgHom.id R S) f ≟ₐ map (AlgHom.id R S) g))
u : T ⊗[R] S ⊗[T] ↥(map (AlgHom.id T T) f ≟ₐ map (AlgHom.id T T) g) →ₐ[T]
  ↥(map (AlgHom.id T (T ⊗[R] S)) (map (AlgHom.id T T) f) ≟ₐ map (AlgHom.id T (T ⊗[R] S)) (map (AlgHom.id T T) g)) :=
  AlgHom.tensorEqualizer T (T ⊗[R] S) (map (AlgHom.id T T) f) (map (AlgHom.id T T) g)
heq :
  map (AlgHom.id T T) (AlgHom.tensorEqualizer R S f g) =
    (↑eRHS.symm).comp ((AlgHom.tensorEqualizer T (T ⊗[R] S) (map (AlgHom.id T T) f) (map (AlgHom.id T T) g)).comp ↑eLHS)
H :
  Function.Bijective
    ⇑((↑eRHS.symm).comp
        ((AlgHom.tensorEqualizer T (T ⊗[R] S) (map (AlgHom.id T T) f) (map (AlgHom.id T T) g)).comp ↑eLHS))
⊢ Function.Bijective (⇑(AlgHom.tensorEqualizer T (T ⊗[R] S) (map (AlgHom.id T T) f) (map (AlgHom.id T T) g)) ∘ ⇑eLHS)
